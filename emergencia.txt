void MakeInvariantMassHist(const std::vector<fastjet::PseudoJet> &charm_jets, const std::vector<fastjet::PseudoJet> &strange_jets, TH1F* invariantMass)
{
    for (const fastjet::PseudoJet &charm_jet : charm_jets)
    {
        TLorentzVector vc(charm_jet.px(), charm_jet.py(), charm_jet.pz(), charm_jet.E());

        for (const fastjet::PseudoJet &strange_jet : strange_jets)
        {
            TLorentzVector vs(strange_jet.px(), strange_jet.py(), strange_jet.pz(), strange_jet.E());

            TLorentzVector vW = vc + vs;
            invariantMass->Fill(vW.M());
        }
    }
}

class JetInfo : public fastjet::PseudoJet::UserInfoBase
{
    public:
        JetInfo(const Float_t& vx, const Float_t& vy, const Float_t& vz) : fpVx(vx), fpVy(vy), fpVz(vz) {}

        void setVx(Float_t vx) { fpVx = vx; }
        Float_t getVx() const { return fpVx; }

        void setVy(Float_t vy) { fpVy = vy; }
        Float_t getVy() const { return fpVy; }

        void setVz(Float_t vz) { fpVz = vz; }
        Float_t getVz() const { return fpVz; }

    private:
        Float_t fpVx;
        Float_t fpVy;
        Float_t fpVz;
};

void invariantMass_classicalReconstruction(const char* fileName)
{
    gSystem->Load("libEG");
    gSystem->Load("libEGPythia8");

    //---------------------------------------------------------------------------------------------------------
    // Initialization of variables
    //---------------------------------------------------------------------------------------------------------

    Float_t fpPt, fpEta, fpPhi, fpE, fpPx, fpPy, fpPz;
    Float_t jetPt, jetEta, jetNConst;
    Float_t vertex_x, vertex_y, vertex_z;


    //---------------------------------------------------------------------------------------------------------
    // Initialization of histograms
    //---------------------------------------------------------------------------------------------------------

    TH1F *combined_cuts_invariantMass = new TH1F("combined_cuts_invariant_mass", "W^{+-} invariant mass spectrum with combined best cuts [GeV/c^{2}]", 100, 0, 120);

    //---------------------------------------------------------------------------------------------------------
    // Initializations and FastJet configurations:
    //---------------------------------------------------------------------------------------------------------

    Float_t jetR = 0.7;

    fastjet::JetDefinition jet_def(fastjet::antikt_algorithm, jetR);

    std::vector<fastjet::PseudoJet> particles_fastjet;
    std::vector<fastjet::PseudoJet> jets;

    //---------------------------------------------------------------------------------------------------------
    // Initialization of the .root file and TTrees
    //---------------------------------------------------------------------------------------------------------

    TFile *file = TFile::Open(fileName, "READ");
    TTree *ttree = dynamic_cast<TTree *>(file->Get("W decay TTree 2"));

    TClonesArray *jets_array = new TClonesArray("MyJet");
    TClonesArray *quarks = new TClonesArray("MyQuark");

    ttree->SetBranchAddress("jets_array", &jets_array);
    ttree->SetBranchAddress("quarks", &quarks);

    Long64_t ne = ttree->GetEntries();

    const Float_t charm_pTCut = 23.5;   const Int_t charm_NConsCut = 5, charm_NRhoCut = 2;
    const Float_t strange_pTCut = 35.1; const Int_t strange_NConsCut = 4, strange_NRhoCut = 0;

    Long64_t n_charm = 0, n_strange = 0, n_ambiguous = 0, n_rejected = 0;

    //---------------------------------------------------------------------------------------------------------
    // Event loop equivalent
    //---------------------------------------------------------------------------------------------------------

    for (Long64_t ni = 0; ni < ne; ni++)
    {
        ttree->GetEntry(ni);

        std::vector<fastjet::PseudoJet> charm_jets;
        std::vector<fastjet::PseudoJet> strange_jets;


        //---------------------------------------------------------------------------------------------------------
        // Particle loop equivalent
        //---------------------------------------------------------------------------------------------------------

        for (Int_t nj = 0; nj < jets_array->GetEntries(); nj++)
        {
            MyJet *fp = static_cast<MyJet *>(jets_array->At(nj));

            fpPx = fp->fPx;
            fpPy = fp->fPy;
            fpPz = fp->fPz;
            fpE = fp->fE;
            vertex_x = fp->fVx;
            vertex_y = fp->fVy;
            vertex_z = fp->fVz;

            fastjet::PseudoJet particle(fpPx, fpPy, fpPz, fpE);

            JetInfo* jetInfo = new JetInfo(vertex_x, vertex_y, vertex_z);
            particle.set_user_info(jetInfo);

            particles_fastjet.push_back(particle);
        }

        fastjet::ClusterSequence clusterSeq(particles_fastjet, jet_def);
        jets = clusterSeq.inclusive_jets();

        for (const fastjet::PseudoJet& jet : jets)
        {
            //---------------------------------------------------------------------------------------------------------
            // Collecting jet's general information
            //---------------------------------------------------------------------------------------------------------

            jetPt = jet.pt();
            if (jetPt < 10) continue;

            jetEta = jet.eta();

            Float_t absEta = TMath::Abs(jetEta);
            if (absEta > 1.3) continue;

            jetNConst = jet.constituents().size();

            Int_t nRho = 0;
            Float_t rhoLowerBound = 0.01;
            Float_t rhoUpperBound = 2;

            for (const fastjet::PseudoJet &constituent : jet.constituents())
            {
                Float_t vx = constituent.user_info<JetInfo>().getVx();
                Float_t vy = constituent.user_info<JetInfo>().getVy();

                Double_t Rho = TMath::Sqrt(pow(vx, 2) + pow(vy, 2));

                if (Rho >= rhoLowerBound && Rho < rhoUpperBound)
                {
                    nRho++;
                }
            }

            bool passC = (jetPt > charm_pTCut   && jetNConst > charm_NConsCut   && nRho > charm_NRhoCut);
            bool passS = (jetPt > strange_pTCut && jetNConst > strange_NConsCut && nRho > strange_NRhoCut);

            if (passC && !passS)      { charm_jets.push_back(jet);   n_charm++; }
            else if (passS && !passC) { strange_jets.push_back(jet); n_strange++; }
            else if (passC && passS)  { n_ambiguous++; }
            else                      { n_rejected++; }
        }

        //---------------------------------------------------------------------------------------------------------
        // Invariant mass reconstruction
        //---------------------------------------------------------------------------------------------------------

        MakeInvariantMassHist(charm_jets, strange_jets, combined_cuts_invariantMass);

        particles_fastjet.clear();
        jets.clear();
        jets_array->Clear();
        quarks->Clear();
    }

    Long64_t n_pass_any = n_charm + n_strange + n_ambiguous;   // jatos únicos que passam em pelo menos um corte
    Double_t ambiguous_pct = (n_pass_any > 0) ? 100.0 * n_ambiguous / n_pass_any : 0.0;

    std::cout   << "Charm inequivoco:  " << n_charm << "\n"
                << "Strange inequivoco:" << n_strange << "\n"
                << "Ambiguos:          " << n_ambiguous << "\n"
                << "Ambiguos / (passam em algum corte): " << ambiguous_pct << "%" << std::endl;

    //---------------------------------------------------------------------------------------------------------
    // Plotting histograms
    //---------------------------------------------------------------------------------------------------------

    gStyle->SetTextFont(42);

    TF1 *f1 = new TF1("f1","[0]+[1]*exp(-(x-[2])^2/(2*[3]))", 50, 110);
    f1->SetParameters(30.0, 70.0, 80.0, 7.0);
    combined_cuts_invariantMass->Fit(f1, "ESR");

    TCanvas *c1 = new TCanvas("c1", "Invariant mass distribution with combined cuts", 2500, 2500);
    c1->Divide(1, 1);

    c1->cd(1);
    combined_cuts_invariantMass->SetTitle("Boson W^{#pm} classical cuts invariant mass spectrum");
    combined_cuts_invariantMass->SetTitleSize(0.04);
    combined_cuts_invariantMass->GetXaxis()->SetTitle("Mass [GeV/c^{2}]");
    combined_cuts_invariantMass->GetYaxis()->SetTitle("Frequency");
    combined_cuts_invariantMass->SetTitleSize(0.04, "XY");
    combined_cuts_invariantMass->SetLineWidth(2);
    combined_cuts_invariantMass->DrawCopy();

    TLegend *leg1 = new TLegend(0.60, 0.70, 0.88, 0.88);
    leg1->AddEntry(f1, "Fit: p_{0}+p_{1}#kern[0.1]{exp}#left[#frac{-(x-p_{2})^{2}}{2p_{3}^{2}}#right]", "l");
    leg1->SetTextSize(0.04);
    leg1->Draw();

    file->Close();
}















//===========================================================================================================================================================================
// Cuts definition:
// 3D grid search over pT, N_const and N_rho thresholds, determining the cut combination
// that maximizes purity subject to a minimum efficiency requirement.
//===========================================================================================================================================================================

#include <string>
#include <vector>
#include <iostream>
#include "TFile.h"
#include "TTree.h"
#include "TSystem.h"
#include "TH2F.h"
#include "TGraph.h"
#include "TCanvas.h"
#include "TLegend.h"

void cuts_definition(const char* strange_file, std::string switch_string)
{
    gSystem->Load("libEG");
    gSystem->Load("libEGPythia8");

    //---------------------------------------------------------------------------------------------------------
    // Data loading
    //---------------------------------------------------------------------------------------------------------

    Float_t pT, nConst, nRho, eventID;
    TTree *signalTree;
    TTree *backgroundTree;

    std::string sfx;

    if (switch_string == "strange")
    {
        sfx = "_s";
    }
    else if (switch_string == "charm")
    {
        sfx = "_c";
    }

    TFile *file = TFile::Open(strange_file, "READ");

    signalTree = dynamic_cast<TTree *>(file->Get(("SignalTree" + sfx).c_str()));
    signalTree->SetBranchAddress(("pT" + sfx).c_str(), &pT);
    signalTree->SetBranchAddress(("nConst" + sfx).c_str(), &nConst);
    signalTree->SetBranchAddress(("nRho" + sfx).c_str(), &nRho);
    signalTree->SetBranchAddress(("eventID" + sfx).c_str(), &eventID);

    backgroundTree = dynamic_cast<TTree *>(file->Get(("BackgroundTree" + sfx).c_str()));
    backgroundTree->SetBranchAddress(("pT" + sfx).c_str(), &pT);
    backgroundTree->SetBranchAddress(("nConst" + sfx).c_str(), &nConst);
    backgroundTree->SetBranchAddress(("nRho" + sfx).c_str(), &nRho);
    backgroundTree->SetBranchAddress(("eventID" + sfx).c_str(), &eventID);

    Long64_t ne_signal = signalTree->GetEntries();
    Long64_t ne_background = backgroundTree->GetEntries();

    std::vector<Float_t> signal_pT, signal_nConst, signal_nRho;
    for (Long64_t i = 0; i < ne_signal; i++)
    {
        signalTree->GetEntry(i);
        signal_pT.push_back(pT);
        signal_nConst.push_back(nConst);
        signal_nRho.push_back(nRho);
    }

    std::vector<Float_t> background_pT, background_nConst, background_nRho;
    for (Long64_t i = 0; i < ne_background; i++)
    {
        backgroundTree->GetEntry(i);
        background_pT.push_back(pT);
        background_nConst.push_back(nConst);
        background_nRho.push_back(nRho);
    }

    //---------------------------------------------------------------------------------------------------------
    // Grid definition
    //---------------------------------------------------------------------------------------------------------

    Float_t pT_min = 10, pT_max = 40, pT_step = 0.1;
    Float_t nConst_min = 0, nConst_max = 30, nConst_step = 1;
    Float_t nRho_min = 0, nRho_max = 10, nRho_step = 1;

    Int_t n_pT = (Int_t)((pT_max - pT_min) / pT_step) + 1;
    Int_t n_nConst = (Int_t)((nConst_max - nConst_min) / nConst_step) + 1;
    Int_t n_nRho = (Int_t)((nRho_max - nRho_min) / nRho_step) + 1;

    std::vector<std::vector<std::vector<Float_t>>> purityGrid(n_pT, std::vector<std::vector<Float_t>>(n_nConst, std::vector<Float_t>(n_nRho, 0)));
    std::vector<std::vector<std::vector<Float_t>>> efficiencyGrid(n_pT, std::vector<std::vector<Float_t>>(n_nConst, std::vector<Float_t>(n_nRho, 0)));

    Float_t bestPurity = -1, bestEfficiency = 0;
    Float_t best_pT_thr = 0, best_nConst_thr = 0, best_nRho_thr = 0;
    Int_t best_i_pT = 0, best_i_nConst = 0, best_i_nRho = 0;

    Float_t minEfficiency;

    if (switch_string == "strange")
    {
        minEfficiency = 0.4;
    }
    else if (switch_string == "charm")
    {
        minEfficiency = 0.6;
    }

    //---------------------------------------------------------------------------------------------------------
    // 3D grid scan
    //---------------------------------------------------------------------------------------------------------

    for (Int_t i_pT = 0; i_pT < n_pT; i_pT++)
    {
        Float_t pT_thr = pT_min + i_pT * pT_step;

        for (Int_t i_nConst = 0; i_nConst < n_nConst; i_nConst++)
        {
            Float_t nConst_thr = nConst_min + i_nConst * nConst_step;

            for (Int_t i_nRho = 0; i_nRho < n_nRho; i_nRho++)
            {
                Float_t nRho_thr = nRho_min + i_nRho * nRho_step;

                Int_t TP = 0, FN = 0, FP = 0, TN = 0;

                for (Long64_t i = 0; i < (Long64_t)signal_pT.size(); i++)
                {
                    if (signal_pT[i] >= pT_thr && signal_nConst[i] >= nConst_thr && signal_nRho[i] >= nRho_thr)
                        TP++;
                    else
                        FN++;
                }

                for (Long64_t i = 0; i < (Long64_t)background_pT.size(); i++)
                {
                    if (background_pT[i] >= pT_thr && background_nConst[i] >= nConst_thr && background_nRho[i] >= nRho_thr)
                        FP++;
                    else
                        TN++;
                }

                // For each combination of cuts, we calculate both the purity and the efficiency

                Float_t purity = (TP + FP > 0) ? (Float_t)TP / (TP + FP) : 0;
                Float_t efficiency = (TP + FN > 0) ? (Float_t)TP / (TP + FN) : 0;

                purityGrid[i_pT][i_nConst][i_nRho] = purity;
                efficiencyGrid[i_pT][i_nConst][i_nRho] = efficiency;

                // Keep the highest purity among combinations satisfying the minimum efficiency

                // For strange jets, the best purity (and I say this based on the previous ML analysis) could happen to be at a lower minimim efficiency than 0.5, the one we used for chamed jets
                // Thus, let me abuse once again the switch string to set a lower minimum for strange jets

                if (efficiency >= minEfficiency && purity > bestPurity)
                {
                    bestPurity = purity;
                    bestEfficiency = efficiency;
                    best_pT_thr = pT_thr;
                    best_nConst_thr = nConst_thr;
                    best_nRho_thr = nRho_thr;
                    best_i_pT = i_pT;
                    best_i_nConst = i_nConst;
                    best_i_nRho = i_nRho;
                }
            }
        }
    }

    std::cout << "Best cut combination (" << switch_string << " jets):" << std::endl;
    std::cout << "pT >= " << best_pT_thr << " GeV/c, nConst >= " << best_nConst_thr << ", nRho >= " << best_nRho_thr << std::endl;
    std::cout << "Purity = " << bestPurity << ", Efficiency = " << bestEfficiency << std::endl;

    //======================================================================================
    // 1D scans: varying one cut while keeping the other two at their optimal values
    //======================================================================================

    //--------------------------------------------------------------------------------------
    // pT scan: N_const and N_rho fixed at their optimal values
    //--------------------------------------------------------------------------------------

    TH1F *pT_cut_purity = new TH1F("pT_cut_purity", "", n_pT, pT_min, pT_max);
    TH1F *pT_cut_efficiency = new TH1F("pT_cut_efficiency", "", n_pT, pT_min, pT_max);

    for (Int_t i_pT = 0; i_pT < n_pT; i_pT++)
    {
        Float_t pT_thr = pT_min + i_pT * pT_step;

        pT_cut_purity->SetBinContent(i_pT, purityGrid[i_pT][best_i_nConst][best_i_nRho]);
        pT_cut_efficiency->SetBinContent(i_pT, efficiencyGrid[i_pT][best_i_nConst][best_i_nRho]);
    }

    //--------------------------------------------------------------------------------------
    // N_const scan: pT and N_rho fixed at their optimal values
    //--------------------------------------------------------------------------------------

    TH1F *nConst_cut_purity = new TH1F("nConst_cut_purity", "", n_nConst, nConst_min, nConst_max);
    TH1F *nConst_cut_efficiency = new TH1F("nConst_cut_efficiency", "", n_nConst, nConst_min, nConst_max);

    for (Int_t i_nConst = 0; i_nConst < n_nConst; i_nConst++)
    {
        Float_t nConst_thr = nConst_min + i_nConst * nConst_step;

        nConst_cut_purity->SetBinContent(i_nConst, purityGrid[best_i_pT][i_nConst][best_i_nRho]);
        nConst_cut_efficiency->SetBinContent(i_nConst, efficiencyGrid[best_i_pT][i_nConst][best_i_nRho]);
    }

    //--------------------------------------------------------------------------------------
    // N_rho scan: pT and N_const fixed at their optimal values
    //--------------------------------------------------------------------------------------

    TH1F *nRho_cut_purity = new TH1F("nRho_cut_purity", "", n_nRho, nRho_min, nRho_max);
    TH1F *nRho_cut_efficiency = new TH1F("nRho_cut_efficiency", "", n_nRho, nRho_min, nRho_max);

    for (Int_t i_nRho = 0; i_nRho < n_nRho; i_nRho++)
    {
        Float_t nRho_thr = nRho_min + i_nRho * nRho_step;

        nRho_cut_purity->SetBinContent(i_nRho, purityGrid[best_i_pT][best_i_nConst][i_nRho]);
        nRho_cut_efficiency->SetBinContent(i_nRho, efficiencyGrid[best_i_pT][best_i_nConst][i_nRho]);
    }

    //======================================================================================
    // Canvases
    //======================================================================================

    gStyle->SetOptStat(0);
    gStyle->SetTextFont(42);
    gStyle->SetOptStat(0);
    gStyle->SetTitleSize(0.04);
    gStyle->SetTitleSize(0.04, "XY");
    gStyle->SetTextSize(0.04);

    //--------------------------------------------------------------------------------------
    // pT scan
    //--------------------------------------------------------------------------------------

    TCanvas *c1 = new TCanvas("c1", "Purity and efficiency", 2500, 1800);
    c1->Divide(1,1);

    c1->cd(1);
    pT_cut_efficiency->SetTitle(("p_{T} scan (" + switch_string + " jets)").c_str());
    pT_cut_efficiency->SetTitleSize(0.04);
    pT_cut_efficiency->GetXaxis()->SetTitleSize(0.04);
    pT_cut_efficiency->GetYaxis()->SetTitleSize(0.04);
    pT_cut_efficiency->GetXaxis()->SetTitle("p_{T} threshold [GeV/c]");
    pT_cut_efficiency->GetYaxis()->SetTitle("Purity and Efficiency");
    pT_cut_efficiency->SetLineColor(kRed);
    pT_cut_efficiency->SetLineWidth(2);
    pT_cut_efficiency->GetYaxis()->SetRangeUser(std::min(pT_cut_efficiency->GetMinimum(), pT_cut_purity->GetMinimum()), std::max(pT_cut_efficiency->GetMaximum(), pT_cut_purity->GetMaximum())*1.2);
    
    Float_t pT_ymin = std::min(pT_cut_efficiency->GetMinimum(), pT_cut_purity->GetMinimum());
    Float_t pT_ymax = std::max(pT_cut_efficiency->GetMaximum(), pT_cut_purity->GetMaximum()) * 1.2;
    pT_cut_efficiency->GetYaxis()->SetRangeUser(pT_ymin, pT_ymax);
    
    pT_cut_efficiency->DrawCopy("HIST");

    TLine *pT_optimal_line = new TLine(best_pT_thr, pT_ymin, best_pT_thr, pT_ymax);
    pT_optimal_line->SetLineColor(kBlack);
    pT_optimal_line->SetLineStyle(2);
    pT_optimal_line->SetLineWidth(3);
    pT_optimal_line->Draw("SAME");

    pT_cut_purity->SetLineColor(kBlue);
    pT_cut_purity->SetLineWidth(2);
    pT_cut_purity->DrawCopy("HIST SAME");

    TLegend *legend1 = new TLegend(0.60, 0.70, 0.90, 0.90);
    legend1->AddEntry(pT_cut_purity, Form("Purity (N_{const} cut = %.0f, N_{#rho} cut = %.0f)", best_nConst_thr, best_nRho_thr), "l");
    legend1->AddEntry(pT_cut_efficiency, Form("Efficiency (N_{const} cut = %.0f, N_{#rho} cut = %.0f)", best_nConst_thr, best_nRho_thr), "l");
    legend1->AddEntry(pT_optimal_line, Form("Optimal p_{T} cut = %.1f GeV/c", best_pT_thr), "l");
    legend1->Draw();

    TCanvas *c2 = new TCanvas("c2", "Purity and efficiency", 2500, 1800);
    c2->Divide(1,1);

    c2->cd(1);
    nConst_cut_efficiency->SetTitle(("N_{const} scan (" + switch_string + " jets)").c_str());
    nConst_cut_efficiency->GetXaxis()->SetTitle("N_{const} threshold");
    nConst_cut_efficiency->GetYaxis()->SetTitle("Purity and Efficiency");
    nConst_cut_efficiency->SetTitleSize(0.04);
    nConst_cut_efficiency->GetXaxis()->SetTitleSize(0.04);
    nConst_cut_efficiency->GetYaxis()->SetTitleSize(0.04);
    nConst_cut_efficiency->SetLineColor(kRed);
    nConst_cut_efficiency->SetLineWidth(2);
    nConst_cut_efficiency->GetYaxis()->SetRangeUser(std::min(nConst_cut_efficiency->GetMinimum(), nConst_cut_purity->GetMinimum()), std::max(nConst_cut_efficiency->GetMaximum(), nConst_cut_purity->GetMaximum())*1.2);

    Float_t nConst_ymin = std::min(nConst_cut_efficiency->GetMinimum(), nConst_cut_purity->GetMinimum());
    Float_t nConst_ymax = std::max(nConst_cut_efficiency->GetMaximum(), nConst_cut_purity->GetMaximum()) * 1.2;
    nConst_cut_efficiency->GetYaxis()->SetRangeUser(nConst_ymin, nConst_ymax);

    nConst_cut_efficiency->DrawCopy("HIST");

    TLine *nConst_optimal_line = new TLine(best_nConst_thr, nConst_ymin, best_nConst_thr, nConst_ymax);
    nConst_optimal_line->SetLineColor(kBlack);
    nConst_optimal_line->SetLineStyle(2);
    nConst_optimal_line->SetLineWidth(3);
    nConst_optimal_line->Draw("SAME");

    nConst_cut_purity->SetLineColor(kBlue);
    nConst_cut_purity->SetLineWidth(2);
    nConst_cut_purity->DrawCopy("HIST SAME");

    TLegend *legend2 = new TLegend(0.60, 0.70, 0.90, 0.90);
    legend2->AddEntry(nConst_cut_purity, Form("Purity (p_{T} cut = %.1f, N_{#rho} cut = %.0f)", best_pT_thr, best_nRho_thr), "l");
    legend2->AddEntry(nConst_cut_efficiency, Form("Efficiency (p_{T} cut = %.1f, N_{#rho} cut = %.0f)", best_pT_thr, best_nRho_thr), "l");
    legend2->AddEntry(nConst_optimal_line, Form("Optimal N_{const} cut = %.0f", best_nConst_thr), "l"); 
    legend2->Draw();

    TCanvas *c3 = new TCanvas("c3", "Purity and efficiency", 2500, 1800);
    c3->Divide(1,1);

    c3->cd(1);
    nRho_cut_efficiency->SetTitle(("N_{#rho} scan (" + switch_string + " jets)").c_str());
    nRho_cut_efficiency->GetXaxis()->SetTitle("N_{#rho} threshold");
    nRho_cut_efficiency->GetYaxis()->SetTitle("Purity and Efficiency");
    nRho_cut_efficiency->SetTitleSize(0.04);
    nRho_cut_efficiency->GetXaxis()->SetTitleSize(0.04);
    nRho_cut_efficiency->GetYaxis()->SetTitleSize(0.04);
    nRho_cut_efficiency->SetLineColor(kRed);
    nRho_cut_efficiency->SetLineWidth(2);
    nRho_cut_efficiency->GetYaxis()->SetRangeUser(std::min(nRho_cut_efficiency->GetMinimum(), nRho_cut_purity->GetMinimum()), std::max(nRho_cut_efficiency->GetMaximum(), nRho_cut_purity->GetMaximum())*1.2);
    
    Float_t nRho_ymin = std::min(nRho_cut_efficiency->GetMinimum(), nRho_cut_purity->GetMinimum());
    Float_t nRho_ymax = std::max(nRho_cut_efficiency->GetMaximum(), nRho_cut_purity->GetMaximum()) * 1.2;
